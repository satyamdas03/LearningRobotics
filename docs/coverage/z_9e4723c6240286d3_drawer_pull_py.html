1"""Drawer-pull problem. 

2 

3A drawer is mounted on a prismatic slide joint with static friction. A constant 

4pulling force is applied via a motor actuator. The question: does the drawer 

5open or jam? 

6 

7Physical concepts: prismatic joints, static friction, actuators, work. 

8""" 

9from __future__ import annotations 

10 

11import numpy as np 

12 

13import mujoco 

14 

15from pibench.core.problem import AnswerType, GroundTruth, Prediction, Problem, Question 

16from pibench.core.registry import register_problem 

17from pibench.utils.articulated import body_id, joint_position 

18from pibench.utils.mjcf import build_xml 

19 

20 

21@register_problem("articulated") 

22class DrawerPull(Problem): 

23 """Does a pulled drawer open or jam against static friction?""" 

24 

25 def _build_scene(self) -> None: 

26 rng = np.random.default_rng(self.seed) 

27 

28 self.drawer_mass = float(rng.uniform(0.5, 2.0)) 

29 self.frictionloss = float(rng.uniform(2.0, 8.0)) 

30 self.applied_force = float(rng.uniform(1.0, 10.0)) 

31 self.open_threshold = 0.05 

32 self.drawer_size = (0.20, 0.15, 0.08) 

33 

34 xml = self._make_xml() 

35 self.model = mujoco.MjModel.from_xml_string(xml) 

36 self.data = mujoco.MjData(self.model) 

37 mujoco.mj_resetData(self.model, self.data) 

38 

39 def _make_xml(self) -> str: 

40 dx, dy, dz = self.drawer_size 

41 worldbody_parts: list[str] = [] 

42 

43 # Cabinet frame: static walls that visually surround the drawer slot. 

44 wall_thickness = 0.02 

45 # Top and bottom plates. 

46 worldbody_parts.append( 

47 f""" <body name=\"cabinet_top\" pos=\"0 0 {dz*2+wall_thickness}\"> 

48 <geom name=\"cabinet_top_geom\" type=\"box\" size=\"{dx} {dy} {wall_thickness}\" rgba=\"0.5 0.5 0.5 1\" /> 

49 </body> 

50""" 

51 ) 

52 worldbody_parts.append( 

53 f""" <body name=\"cabinet_bottom\" pos=\"0 0 {-wall_thickness}\"> 

54 <geom name=\"cabinet_bottom_geom\" type=\"box\" size=\"{dx} {dy} {wall_thickness}\" rgba=\"0.5 0.5 0.5 1\" /> 

55 </body> 

56""" 

57 ) 

58 

59 # Drawer on a prismatic joint along +X. 

60 worldbody_parts.append( 

61 f""" <body name=\"drawer\" pos=\"0 0 {dz}\"> 

62 <joint name=\"drawer_joint\" type=\"slide\" axis=\"1 0 0\" range=\"0 0.5\" frictionloss=\"{self.frictionloss}\" damping=\"0.01\" /> 

63 <inertial pos=\"0 0 0\" mass=\"{self.drawer_mass}\" diaginertia=\"{self.drawer_mass/12*(dy**2+dz**2)} {self.drawer_mass/12*(dx**2+dz**2)} {self.drawer_mass/12*(dx**2+dy**2)}\" /> 

64 <geom name=\"drawer_geom\" type=\"box\" size=\"{dx} {dy} {dz}\" rgba=\"0.7 0.4 0.2 1\" /> 

65 <geom name=\"drawer_handle\" type=\"sphere\" size=\"0.03\" pos=\"{dx+0.05} 0 0\" rgba=\"0.3 0.3 0.3 1\" /> 

66 </body> 

67""" 

68 ) 

69 

70 xml = build_xml("".join(worldbody_parts), timestep=0.002) 

71 actuator = f""" <actuator> 

72 <motor joint=\"drawer_joint\" gear=\"1\" ctrlrange=\"0 {self.applied_force}\" /> 

73 </actuator> 

74""" 

75 xml = xml.replace("</mujoco>", actuator + "</mujoco>") 

76 return xml 

77 

78 def _run_outcome(self) -> str: 

79 if getattr(self, "_outcome", None) is not None: 

80 return self._outcome 

81 

82 # Apply a constant pulling force. 

83 self.data.ctrl[0] = self.applied_force 

84 initial_q = joint_position(self.model, self.data, "drawer_joint") 

85 

86 steps = 1000 

87 max_q = initial_q 

88 for _ in range(steps): 

89 mujoco.mj_step(self.model, self.data) 

90 max_q = max(max_q, joint_position(self.model, self.data, "drawer_joint")) 

91 

92 opened = max_q - initial_q > self.open_threshold 

93 outcome = "yes" if opened else "no" 

94 self._outcome = outcome 

95 self._max_displacement = float(max_q - initial_q) 

96 return outcome 

97 

98 def question(self) -> Question: 

99 return Question( 

100 text=( 

101 f"A {self.drawer_mass:.2f} kg drawer is pulled with a constant force of " 

102 f"{self.applied_force:.2f} N. The slide joint has {self.frictionloss:.2f} N " 

103 f"of static friction. Does the drawer open?" 

104 ), 

105 answer_type=AnswerType.BOOLEAN, 

106 ) 

107 

108 def ground_truth(self) -> GroundTruth: 

109 outcome = self._run_outcome() 

110 return GroundTruth( 

111 answer=outcome, 

112 explanation=( 

113 f"Drawer mass {self.drawer_mass:.2f} kg, applied force {self.applied_force:.2f} N, " 

114 f"static friction {self.frictionloss:.2f} N. " 

115 f"Maximum displacement: {self._max_displacement:.4f} m. " 

116 f"Drawer {'opens' if outcome == 'yes' else 'jams'}." 

117 ), 

118 latent_params={ 

119 "drawer_mass": self.drawer_mass, 

120 "applied_force": self.applied_force, 

121 "frictionloss": self.frictionloss, 

122 "max_displacement": self._max_displacement, 

123 }, 

124 ) 

125 

126 def score(self, prediction: Prediction) -> float: 

127 gt = self.ground_truth().answer 

128 pred = str(prediction.answer).lower() 

129 return 1.0 if pred in (gt, "true", "1") else 0.0
