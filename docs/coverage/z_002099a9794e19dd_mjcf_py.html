5def mjcf_header(timestep: float = 0.002, gravity: str = "0 0 -9.81") -> str: 

6 """Return a standard MuJoCo XML header.""" 

7 return f"""<mujoco model="pibench_scene"> 

8 <compiler angle="radian" /> 

9 <option timestep="{timestep}" gravity="{gravity}" iterations="50" solver="Newton" /> 

10 <visual> 

11 <global azimuth="120" elevation="-20" /> 

12 </visual> 

13"""

26def mjcf_box(name: str, pos: tuple[float, float, float], size: tuple[float, float, float], 

27 rgba: tuple[float, float, float, float] = (0.8, 0.2, 0.2, 1.0), 

28 mass: float | None = None, friction: tuple[float, float, float] | None = None) -> str: 

29 """Return a free-floating box body.""" 

30 friction_attr = "" 

31 if friction is not None: 

32 friction_attr = f' friction="{friction[0]} {friction[1]} {friction[2]}"' 

33 inertial = "" 

34 if mass is not None: 

35 inertial = f'\n <inertial pos="0 0 0" mass="{mass}" diaginertia="{mass/12*(size[1]**2+size[2]**2)} {mass/12*(size[0]**2+size[2]**2)} {mass/12*(size[0]**2+size[1]**2)}" />' 

36 return f""" <body name="{name}" pos="{pos[0]} {pos[1]} {pos[2]}">{inertial} 

37 <freejoint /> 

38 <geom name="{name}_geom" type="box" size="{size[0]} {size[1]} {size[2]}" rgba="{rgba[0]} {rgba[1]} {rgba[2]} {rgba[3]}"{friction_attr} /> 

39 </body> 

40"""

60def mjcf_sphere(name: str, pos: tuple[float, float, float], radius: float, 

61 rgba: tuple[float, float, float, float] = (0.8, 0.2, 0.2, 1.0), 

62 mass: float | None = None, friction: tuple[float, float, float] | None = None) -> str: 

63 """Return a free-floating sphere body.""" 

64 friction_attr = "" 

65 if friction is not None: 

66 friction_attr = f' friction="{friction[0]} {friction[1]} {friction[2]}"' 

67 inertial = "" 

68 if mass is not None: 

69 # Solid sphere inertia: (2/5) * m * r^2. 

70 i = 0.4 * mass * radius * radius 

71 inertial = f'\n <inertial pos="0 0 0" mass="{mass}" diaginertia="{i} {i} {i}" />' 

72 return f""" <body name="{name}" pos="{pos[0]} {pos[1]} {pos[2]}">{inertial} 

73 <freejoint /> 

74 <geom name="{name}_geom" type="sphere" size="{radius}" rgba="{rgba[0]} {rgba[1]} {rgba[2]} {rgba[3]}"{friction_attr} /> 

75 </body> 

76"""
